import pandas as pd import numpy as np # 1. CHARGEMENT DES DONNÉES (Basé sur vos noms de fichiers) # On charge les paramètres personnels (Matricule, Diplômes, Spécialité) df_perso = pd.read_csv('TABLEAU DE BORD SUIVIE PERSONNELS DU CI.xlsx - PARAMETRES PERSONNELS.csv') # On charge les données médicales (Poids, Taille, IMC, Tension) df_sante = pd.read_csv('TABLEAU DE BORD SUIVIE PERSONNELS DU CI.xlsx - INFORMATIONS MEDICALES.csv') # On charge les évaluations (COVAPI, Tir, Morale) df_eval = pd.read_csv('TABLEAU DE BORD SUIVIE PERSONNELS DU CI.xlsx - EVALUATION PERSONNEL.csv') # 2. NETTOYAGE ET FUSION (Le "Merge") # On utilise le NOM comme clé de liaison pour regrouper toutes les informations df_complet = pd.merge(df_perso, df_sante[['NOM', 'POIDS Kg', 'TAILLE m', 'TA', 'GLYCEMIE']], on='NOM', how='left') df_complet = pd.merge(df_complet, df_eval[['NOM', 'COVAPI', 'VALEUR MORALE', 'NOTE TIR']], on='NOM', how='left') # 3. CALCULS AUTOMATISÉS (Intelligence du code) def calculer_imc(row): try: if row['TAILLE m'] > 0: return round(row['POIDS Kg'] / (row['TAILLE m']**2), 2) except: return np.nan # Recalcul de l'IMC pour vérifier les données du fichier df_complet['IMC_CALCULE'] = df_complet.apply(calculer_imc, axis=1) # Calcul d'une Moyenne Générale pondérée (Exemple: Physique 30%, Tir 40%, Morale 30%) df_complet['SCORE_GLOBAL'] = ( df_complet['COVAPI'].fillna(0) * 0.3 + df_complet['NOTE TIR'].fillna(0) * 0.4 + df_complet['VALEUR MORALE'].fillna(0) * 0.3 ) # 4. EXEMPLE DE SORTIE : EXTRAIRE LES PROFILS À RISQUE OU D'ÉLITE # Liste des personnels ayant un IMC > 25 (Surpoids) profils_imc_eleve = df_complet[df_complet['IMC_CALCULE'] > 25][['NOM', 'GRADE', 'IMC_CALCULE']] # Liste des meilleurs tireurs (Note > 15) tireurs_elite = df_complet[df_complet['NOTE TIR'] >= 16][['NOM', 'GRADE', 'NOTE TIR']] # 5. EXPORTATION VERS UN NOUVEAU FICHIER CENTRALISÉ df_complet.to_excel('BASE_DE_DONNEES_INTEGREE_CI.xlsx', index=False) print("Analyse terminée. Fichier centralisé généré avec succès.")
